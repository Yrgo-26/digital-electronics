% Digitalteknik - the book.
%
% Built with LuaLaTeX; see the Makefile. The design is in dtbook.sty, the front matter in front/,
% each chapter in chapters/NN/, one file per section, mirroring the appendices of the lecture it is
% typeset from, and the back matter in back/: the three lab guides, the Microchip Studio guide and
% the instruction sheet from info/, the practice exam from exam/, and the solutions to the
% exercises that are not programs.
%
% `make part PART=04` builds the front matter and one chapter alone, with its solutions (or
% PART=lab1, lab2, lab3, studio, avr, exam or answers), for checking one piece quickly: \onlypart
% names it, and every piece skipped still steps the chapter counter, so the one built keeps its
% number.
\documentclass[10pt, twoside, openright]{book}
\usepackage{dtbook}

% Nothing named means the whole book. \def rather than \providecommand, which would make it
% \long and so never equal to \empty.
\ifdefined\onlypart\else\def\onlypart{}\fi
\makeatletter
% \bookpart{key}{file}: input file, unless one other part was asked for.
\newcommand{\bookpart}[2]{%
  \ifx\onlypart\empty\input{#2}\else
    \def\@tempa{#1}\ifx\@tempa\onlypart\input{#2}\else\refstepcounter{chapter}\fi
  \fi}
% \bookanswers{key}{file}: one chapter's solutions, input with the whole book, with the solutions
% alone, or with the chapter they answer.
\newcommand{\bookanswers}[2]{%
  \ifx\onlypart\empty\input{#2}\else
    \def\@tempa{#1}\ifx\@tempa\onlypart\input{#2}\else
      \def\@tempa{answers}\ifx\@tempa\onlypart\input{#2}\fi\fi
  \fi}
\makeatother

\bookpart{avr}{back/avr}

\bookpart{exam}{back/exam/practice}
\bookpart{exam}{back/exam/practice_answers}
\input{back/answers/answers}

\end{document}
